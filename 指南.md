- 速度：0.12秒/页平均
- 120秒预算：500-1000页

```python
import pymupdf
import pymupdf4llm
from docx import Document
from pathlib import Path
import json

class DocumentAuditProcessor:
    """文档审核处理器"""
    
    def __init__(self, output_dir="processed"):
        self.output_dir = Path(output_dir)
        self.output_dir.mkdir(exist_ok=True)
    
    def process_document(self, file_path):
        """主入口 - 根据格式路由"""
        file_path = Path(file_path)
        suffix = file_path.suffix.lower()
        
        if suffix == '.pdf':
            return self.process_pdf(file_path)
        elif suffix == '.docx':
            return self.process_docx(file_path)
        else:
            raise ValueError(f"不支持的格式: {suffix}")
    
    def process_pdf(self, pdf_path):
        """处理PDF文档，提取结构化信息"""
        doc = pymupdf.open(pdf_path)
        
        # 提取为Markdown格式（适合VLM输入）
        md_text = pymupdf4llm.to_markdown(
            doc=str(pdf_path),
            page_chunks=True,
            write_images=True,
            image_path=str(self.output_dir / "images"),
            embed_images=False
        )
        
        # 提取结构化元数据
        structure = []
        for page_num, page in enumerate(doc):
            page_info = {
                'page': page_num + 1,
                'paragraphs': [],
                'tables': [],
                'images': []
            }
            
            # 提取文本块及位置
            text_dict = page.get_text("dict")
            para_idx = 0
            for block in text_dict["blocks"]:
                if block["type"] == 0:  # 文本
                    text = " ".join(
                        s["text"] for line in block["lines"] 
                        for s in line["spans"]
                    )
                    page_info['paragraphs'].append({
                        'index': para_idx,
                        'bbox': block["bbox"],
                        'text': text
                    })
                    para_idx += 1
            
            # 提取表格
            tables = page.find_tables()
            for idx, table in enumerate(tables):
                page_info['tables'].append({
                    'index': idx,
                    'bbox': table.bbox,
                    'data': table.extract()
                })
            
            structure.append(page_info)
        
        doc.close()
        return {
            'markdown': md_text,
            'structure': structure,
            'format': 'pdf',
            'page_count': len(structure)
        }
```

### 1.2 文档解析为VLM输入

**推荐方案：混合多模态** - 结合Markdown文本 + 页面图片（150 DPI），性能0.15秒/页，可在120秒处理约400页。

```python
def prepare_for_vlm(pdf_path, dpi=150):
    """为VLM处理准备文档"""
    doc = pymupdf.open(pdf_path)
    
    # 提取Markdown文本
    md_text = pymupdf4llm.to_markdown(doc=pdf_path, page_chunks=True)
    
    # 提取页面图片
    page_images = []
    for page_num, page in enumerate(doc):
        mat = pymupdf.Matrix(dpi/72, dpi/72)
        pix = page.get_pixmap(matrix=mat, alpha=False)
        
        page_images.append({
            'page': page_num + 1,
            'image_bytes': pix.tobytes("png"),
            'width': pix.width,
            'height': pix.height
        })
        pix = None
    
    doc.close()
    return {'text': md_text, 'images': page_images}
```

### 1.3 结构化信息提取

提取章节、表格、图片、页码和段落索引，用于后续精确定位批注位置。

```python
def extract_all_structure(pdf_path):
    """提取完整文档结构"""
    doc = pymupdf.open(pdf_path)
    
    # 从目录提取章节
    toc = doc.get_toc()
    chapters = [{'level': lvl, 'title': title, 'page': page} 
                for lvl, title, page in toc] if toc else []
    
    # 提取所有表格
    all_tables = []
    for page_num, page in enumerate(doc):
        tables = page.find_tables()
        for idx, table in enumerate(tables):
            all_tables.append({
                'page': page_num + 1,
                'table_index': idx,
                'data': table.extract()
            })
    
    # 提取带索引的段落
    paragraphs = []
    para_idx = 0
    for page_num, page in enumerate(doc):
        text_dict = page.get_text("dict")
        for block in text_dict["blocks"]:
            if block["type"] == 0:
                text = " ".join(s["text"] for line in block["lines"] 
                               for s in line["spans"])
                paragraphs.append({
                    'index': para_idx,
                    'page': page_num + 1,
                    'text': text,
                    'bbox': block["bbox"]
                })
                para_idx += 1
    
    doc.close()
    return {
        'chapters': chapters,
        'tables': all_tables,
        'paragraphs': paragraphs
    }
```

规则动态注入（支持50-200条规则）：

```python
import yaml

def load_and_inject_rules(config_path: str) -> str:
    """加载YAML并转换为层级文本"""
    with open(config_path) as f:
        rules = yaml.safe_load(f)
    
    output = []
    for scenario in rules['audit_scenarios']:
        output.append(f"# {scenario['name']}\n")
        for dimension in scenario['dimensions']:
            output.append(f"## {dimension['name']} (权重: {dimension['weight']:.0%})\n")
            for item in dimension['items']:
                output.append(f"### {item['name']}\n")
                for cp in item['checkpoints']:
                    output.append(
                        f"**[{cp['checkpoint_id']}]** {cp['description']}\n"
                        f"- 严重程度: {cp['severity']}\n"
                        f"- 最大扣分: {cp['max_deduction']} 分\n"
                    )
    
    return "\n".join(output)
```

### 4.2 PDF批注

**推荐：PyMuPDF**（性能最佳，功能最全）

```python
import pymupdf

def annotate_pdf_findings(pdf_path, findings, output_path):
    """在PDF中添加批注"""
    doc = pymupdf.open(pdf_path)
    
    colors = {
        'critical': (1, 0, 0),
        'high': (1, 0.5, 0),
        'medium': (1, 1, 0),
        'low': (0.5, 1, 0)
    }
    
    for finding in findings:
        page = doc[finding['page'] - 1]
        
        # 文本搜索定位
        rects = page.search_for(finding['search_text'])
        
        if rects:
            color = colors.get(finding['severity'], (1, 1, 0))
            
            # 添加高亮
            highlight = page.add_highlight_annot(rects)
            highlight.set_colors(stroke=color)
            highlight.set_opacity(0.5)
            highlight.set_info(
                title=f"问题 #{finding['id']}",
                content=finding['comment']
            )
            highlight.update()
            
            # 添加便签
            note = page.add_text_annot(rects[0].top_left, finding['comment'])
            note.set_colors(stroke=color)
            note.update()
    
    doc.save(output_path)
    doc.close()
```

模糊匹配（应对OCR误差）：

```python
from fuzzywuzzy import fuzz

def fuzzy_annotate_pdf(pdf_path, findings, output_path, threshold=80):
    """使用模糊匹配添加批注"""
    doc = pymupdf.open(pdf_path)
    
    for finding in findings:
        page = doc[finding['page'] - 1]
        blocks = page.get_text("blocks")
        
        best_match = None
        best_score = 0
        
        for x0, y0, x1, y1, text, *_ in blocks:
            similarity = fuzz.ratio(
                finding['search_text'].lower(), 
                text.lower()
            )
            
            if similarity > best_score and similarity >= threshold:
                best_score = similarity
                best_match = pymupdf.Rect(x0, y0, x1, y1)
        
        if best_match:
            h = page.add_highlight_annot(best_match)
            h.set_info(content=finding['comment'])
            h.update()
    
    doc.save(output_path)
    doc.close()
```

```python
def generate_audit_report(findings_data, output_path):
```

### 5.3 异步并行处理

```python
import asyncio
from concurrent.futures import ProcessPoolExecutor

class AsyncDocumentPipeline:
    """生产级异步Pipeline"""
    
    def __init__(self, max_workers=8):
        self.cpu_executor = ProcessPoolExecutor(max_workers=max_workers)
    
    async def stage_conversion(self, doc_data):
        """转换阶段（CPU密集，使用进程池）"""
        loop = asyncio.get_event_loop()
        images = await loop.run_in_executor(
            self.cpu_executor,
            convert_pdf_parallel,
            doc_data
        )
        return images
    
    async def stage_vlm_batch(self, batch):
        """VLM批处理（GPU密集）"""
        prompts = [prepare_vlm_prompt(doc) for doc in batch]
        outputs = await self.llm.generate_async(prompts, self.sampling_params)
        return outputs
    
    async def run_pipeline(self, documents, timeout=120):
        """运行完整pipeline"""
        import time
        start_time = time.time()
        
        # 启动workers
        workers = [
            *[asyncio.create_task(self.conversion_worker()) for _ in range(4)],
            *[asyncio.create_task(self.preprocessing_worker()) for _ in range(4)],
            *[asyncio.create_task(self.vlm_batch_worker()) for _ in range(2)],
            *[asyncio.create_task(self.output_worker()) for _ in range(4)],
        ]
        
        try:
            await asyncio.wait_for(
                self.wait_for_completion(),
                timeout=timeout - (time.time() - start_time)
            )
        except asyncio.TimeoutError:
            print(f"Pipeline超时")
        
        return await self.collect_results()
```

## 10. 常见问题与解决方案

### Q1: 如何处理超过120秒的文档？
**A**: 实施分页策略，限制最大页数（MAX_PAGES=20），或使用更激进的token限制（max_tokens=256）。

### Q2: AGPL许可证是否影响商业使用？
**A**: PyMuPDF的AGPL许可证要求开源或购买商业许可。替代方案：pypdf（BSD许可，功能较少）。

### Q3: 如何提升VLM推理速度？
**A**: 
- 设置`--num-scheduler-steps 10`（+2.7倍）
- 增加`max_num_batched_tokens=16384`（+28%）
- 降低温度至0.1（+10-15%）
- 限制max_tokens至512（+20-30%）

### Q4: 批注定位不准确怎么办？
**A**: 使用混合定位策略：
1. 优先文本精确搜索
2. 后备模糊匹配（threshold=80）
3. 最后使用坐标定位（如VLM提供）

### Q5: 如何处理格式损坏的文档？
**A**: 实施多解析器后备策略：
```python
parsers = ['pymupdf', 'pypdf2', 'pdfplumber', 'pikepdf']
for parser in parsers:

        return parse_with(parser, pdf_path)
    except:
        continue
```
